# Generate the ROS topic/node constants as a named module interface.
set(ros_info_module ${CMAKE_CURRENT_BINARY_DIR}/ros_info.cppm)
add_custom_command(
    OUTPUT ${ros_info_module}
    COMMAND
        python3 ${CMAKE_SOURCE_DIR}/scripts/gen_ros_info.py
        --cpp-output ${ros_info_module}
    DEPENDS
        ${CMAKE_SOURCE_DIR}/scripts/gen_ros_info.py
        ${CMAKE_SOURCE_DIR}/ros_info.yaml
    BYPRODUCTS ${CMAKE_SOURCE_DIR}/launch/ros_info.py
)
set_source_files_properties(${ros_info_module} PROPERTIES GENERATED TRUE)
add_custom_target(ros_info_module DEPENDS ${ros_info_module})
set(AUTOGEN_TARGETS "ros_info_module" CACHE INTERNAL "Set autogenerated targets")

set(module network_common)
set(srcs)
set(module_srcs
    ${ros_info_module}
    ${CMAKE_CURRENT_LIST_DIR}/shared_constants.cppm
    ${CMAKE_CURRENT_LIST_DIR}/../utils/utils.cppm
)
set(link_libs)
set(inc_dirs)
set(compile_defs)
make_lib(
    ${module}
    "${srcs}"
    "${module_srcs}"
    "${link_libs}"
    "${inc_dirs}"
    "${compile_defs}"
)
